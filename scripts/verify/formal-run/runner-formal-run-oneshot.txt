cat > ~/hbr-formal-run.sh <<'WRAP_EOF'
#!/usr/bin/env bash
# P1-HBR-01 IF-09 Runner 정식 시험 wrapper v2 (AC-02). 본체: infra-harbor-trivy main 6edf816 (tree 25b8195fe1c1), 판정 로직은 변경하지 않는다.
# 종료 코드: 0=PASS, 1=본체 FAIL, 2=ERROR(본체 ERROR 또는 wrapper/증적 처리 실패). PASS는 모든 조건이 참일 때만 출력한다.
set -u
umask 077
EXPECT_SCRIPT_SHA=1305e9affe70132bcc570a646369834dd1ce237570a4c7537bbead6da6b6fe1e
BASE_IMAGE_REF='alpine@sha256:c053b5923901c5057d17732b5b42421b05215f963078a05c742c3d639b7a4214'
CA_SHA=20ddfa5e92c346aaf46fc30179fbc0b069e256effd2615a9f4308a8fe719053f
SECRET_SRC=/dev/tty
die() { echo "ERROR: $*" >&2; exit 2; }
for c in podman sha256sum base64 tee grep head tr date mktemp od cut hostname whoami rm mkdir; do command -v "$c" >/dev/null 2>&1 || die "필수 명령 없음: $c"; done
# 사전 주입된 비밀·입력 거부 (Formal Run은 터미널 입력만 허용)
[ -z "${RUNNER_ROBOT_SECRET+x}" ] || die "RUNNER_ROBOT_SECRET가 이미 설정되어 있습니다. 사전 주입은 허용하지 않습니다(터미널 입력만)."
[ -z "${WRONG_ROBOT_SECRET+x}" ] || die "WRONG_ROBOT_SECRET 사전 설정은 허용하지 않습니다."
[ -z "${BASE_IMAGE_REF_OVERRIDE+x}" ] || die "BASE_IMAGE_REF 변경은 허용하지 않습니다(고정 digest만)."
WRAPPER_SHA256="$(sha256sum "$0" | cut -d' ' -f1)" || die "wrapper 해시 계산 실패"
# 실행마다 유일한 run 디렉터리 (mkdir -p 금지: 이미 있으면 중단)
NONCE="$(head -c 4 /dev/urandom | od -An -tx1 | tr -d ' \n')"
RUN_ID="hbr-$(date +%Y%m%d-%H%M%S)-${NONCE}"
# AC-03 로더(load-review-run-env.sh) 규칙과 동일하게 Runtime 이전에 검증한다. timestamp/nonce 생성 실패·빈 값은 여기서 STOP.
[[ "$RUN_ID" =~ ^hbr-[0-9]{8}-[0-9]{6}-[0-9a-f]{8}$ ]] || die "RUN_ID 형식 오류(timestamp/nonce 생성 실패): '$RUN_ID'"
[[ "$RUN_ID" =~ ^[a-z0-9]([a-z0-9-]{0,47}[a-z0-9])?$ ]] || die "RUN_ID가 AC-03 로더 규칙을 위반합니다: '$RUN_ID'"
DIR="$HOME/hbr-review-run-$RUN_ID"
mkdir "$DIR" 2>/dev/null || die "run 디렉터리가 이미 존재하거나 만들 수 없습니다: $DIR (재사용 금지)"
LOG="$DIR/run.log"
cat > "$DIR/runner-push-test.sh" <<'SCRIPT_EOF'
#!/usr/bin/env bash
# P1-HBR-01 / IF-09 — Runner Push/Pull Positive & Negative Test
#
# 종료 코드: 0=PASS, 1=FAIL(기대와 다른 결과), 2=ERROR(원인을 확정할 수 없음 — PASS도 FAIL도 아님)
# 필수 입력(환경변수, 값은 출력하지 않는다):
#   RUNNER_ROBOT_USER / RUNNER_ROBOT_SECRET  Runner 정상 credential
#   WRONG_ROBOT_SECRET                       음성시험용 잘못된 secret
#   BASE_IMAGE_REF                           digest로 고정된 시험 입력 이미지 (예: alpine@sha256:<64 hex>)
set -euo pipefail

HARBOR_REGISTRY="10.1.93.54"
DIGEST_RE='^sha256:[0-9a-f]{64}$'
BASE_REF_RE='^[^@[:space:]]+@sha256:[0-9a-f]{64}$'
AUTH_REJECT_MARKERS='unauthorized|401|authentication required|incorrect username or password|invalid username'
AMBIGUOUS_MARKERS='timeout|TLS|certificate|connection refused|EOF|temporary failure|dial tcp|no such host|network is unreachable'

TEST_DOCKER_CONFIG="$(mktemp -d)"
WORK_DIR="$(mktemp -d)"
export DOCKER_CONFIG="${TEST_DOCKER_CONFIG}"
set +x

TEST_TAG="ifup-$(date -u +%Y%m%dT%H%M%SZ)"
TEST_IMAGE="${HARBOR_REGISTRY}/fds/test:${TEST_TAG}"

cleanup() {
  podman logout "${HARBOR_REGISTRY}" >/dev/null 2>&1 || true
  podman rmi -f "${TEST_IMAGE}" >/dev/null 2>&1 || true
  rm -rf "${TEST_DOCKER_CONFIG}" "${WORK_DIR}"
  unset DOCKER_CONFIG RUNNER_ROBOT_SECRET WRONG_ROBOT_SECRET
}
trap cleanup EXIT

if [ -z "${RUNNER_ROBOT_USER:-}" ] || [ -z "${RUNNER_ROBOT_SECRET:-}" ]; then
  echo "ERROR: RUNNER_ROBOT_USER 또는 RUNNER_ROBOT_SECRET 환경변수가 주입되지 않았습니다."
  exit 2
fi
if [ -z "${WRONG_ROBOT_SECRET:-}" ]; then
  echo "ERROR: WRONG_ROBOT_SECRET 환경변수(음성시험용, 임의 문자열)가 필요합니다."
  exit 2
fi
if [ -z "${BASE_IMAGE_REF:-}" ] || ! printf '%s' "${BASE_IMAGE_REF}" | grep -qE "${BASE_REF_RE}"; then
  echo "ERROR: BASE_IMAGE_REF는 digest 고정 참조(name@sha256:<64 hex>)여야 합니다. moving tag는 허용하지 않습니다."
  exit 2
fi
echo "INFO: base_image=${BASE_IMAGE_REF}"

echo "### 1. Positive: Runner Credential 로그인"
LOGIN_EXIT_CODE=0
printf '%s' "${RUNNER_ROBOT_SECRET}" | \
  podman login "${HARBOR_REGISTRY}" \
    --username "${RUNNER_ROBOT_USER}" \
    --password-stdin || LOGIN_EXIT_CODE=$?
if [ ${LOGIN_EXIT_CODE} -ne 0 ]; then
  echo "FAIL: 정상 credential로도 로그인 실패 — IF-09 Positive 시험 무의미"
  exit 1
fi
echo "OK: 로그인 성공"

echo "### 2. Positive: 이미지 push (manifest digest는 push가 기록한 digestfile에서 읽는다)"
PULL_BASE_EXIT_CODE=0
podman pull "${BASE_IMAGE_REF}" >/dev/null || PULL_BASE_EXIT_CODE=$?
if [ ${PULL_BASE_EXIT_CODE} -ne 0 ]; then
  echo "ERROR: 시험 입력 이미지를 받지 못했습니다 (Exit Code: ${PULL_BASE_EXIT_CODE}) — Harbor 시험 이전 단계 실패"
  exit 2
fi
podman tag "${BASE_IMAGE_REF}" "${TEST_IMAGE}"
DIGEST_FILE="${WORK_DIR}/push.digest"
PUSH_EXIT_CODE=0
PUSH_OUTPUT=$(podman push --digestfile "${DIGEST_FILE}" "${TEST_IMAGE}" 2>&1) || PUSH_EXIT_CODE=$?
if [ ${PUSH_EXIT_CODE} -ne 0 ]; then
  echo "FAIL: Runner Push 실패 (Exit Code: ${PUSH_EXIT_CODE})"
  echo "Output: ${PUSH_OUTPUT}"
  exit 1
fi
PUSHED_DIGEST=""
if [ -s "${DIGEST_FILE}" ]; then
  PUSHED_DIGEST="$(tr -d '[:space:]' < "${DIGEST_FILE}")"
fi
if ! printf '%s' "${PUSHED_DIGEST}" | grep -qE "${DIGEST_RE}"; then
  echo "ERROR: push가 manifest digest를 기록하지 않았거나 형식이 올바르지 않습니다 (DIGEST_UNVERIFIED)"
  exit 2
fi
echo "PASS: Runner Push 성공 image=${TEST_IMAGE} manifest_digest=${PUSHED_DIGEST}"

echo "### 3. Positive: 같은 Runner credential로 Harbor에서 exact digest Pull"
# 로컬에 같은 내용이 남아 있으면 Harbor Pull이 아니므로 입력/시험 이미지를 모두 지운 뒤 받는다.
podman rmi -f "${TEST_IMAGE}" "${BASE_IMAGE_REF}" >/dev/null 2>&1 || true
PULL_REF="${HARBOR_REGISTRY}/fds/test@${PUSHED_DIGEST}"
PULL_EXIT_CODE=0
PULL_OUTPUT=$(podman pull "${PULL_REF}" 2>&1) || PULL_EXIT_CODE=$?
if [ ${PULL_EXIT_CODE} -ne 0 ]; then
  echo "FAIL: Harbor에서 digest Pull 실패 (Exit Code: ${PULL_EXIT_CODE})"
  echo "Output: ${PULL_OUTPUT}"
  exit 1
fi
PULLED_DIGEST="$(podman image inspect --format '{{.Digest}}' "${PULL_REF}" 2>/dev/null | tr -d '[:space:]' || true)"
if [ "${PULLED_DIGEST}" != "${PUSHED_DIGEST}" ]; then
  echo "FAIL: Pull한 이미지 digest(${PULLED_DIGEST:-없음})가 push digest(${PUSHED_DIGEST})와 다름"
  exit 1
fi
echo "PASS: Harbor digest Pull 성공 및 digest 일치 pulled=${PULLED_DIGEST}"
podman rmi -f "${PULL_REF}" >/dev/null 2>&1 || true
podman logout "${HARBOR_REGISTRY}" >/dev/null 2>&1 || true

echo "### 4. Negative: 잘못된 credential 로그인 시도 (인증 거부가 관측돼야 PASS)"
WRONG_DOCKER_CONFIG="$(mktemp -d)"
WRONG_LOGIN_EXIT_CODE=0
WRONG_LOGIN_OUTPUT=$(
  export DOCKER_CONFIG="${WRONG_DOCKER_CONFIG}"
  printf '%s' "${WRONG_ROBOT_SECRET}" | podman login "${HARBOR_REGISTRY}" --username "${RUNNER_ROBOT_USER}" --password-stdin 2>&1
) || WRONG_LOGIN_EXIT_CODE=$?
rm -rf "${WRONG_DOCKER_CONFIG}"

if [ ${WRONG_LOGIN_EXIT_CODE} -eq 0 ]; then
  echo "FAIL: 잘못된 credential로 로그인 성공함 (거부돼야 하는데 안 됨)"
  exit 1
elif printf '%s' "${WRONG_LOGIN_OUTPUT}" | grep -iqE "${AMBIGUOUS_MARKERS}"; then
  echo "ERROR: 로그인 실패 원인에 인증 외 오류(네트워크/TLS) 신호가 섞여 있어 권한 거부로 확정할 수 없음"
  echo "Output: ${WRONG_LOGIN_OUTPUT}"
  exit 2
elif printf '%s' "${WRONG_LOGIN_OUTPUT}" | grep -iqE "${AUTH_REJECT_MARKERS}"; then
  echo "PASS: 잘못된 credential 로그인이 인증 거부로 확인됨"
else
  echo "ERROR: 로그인은 실패했지만 인증 거부 신호(401/unauthorized 등)가 없어 원인을 확정할 수 없음 (UNRESOLVED)"
  echo "Output: ${WRONG_LOGIN_OUTPUT}"
  exit 2
fi

echo "### 완료 — TEST_TAG=${TEST_TAG} MANIFEST_DIGEST=${PUSHED_DIGEST} BASE_IMAGE_REF=${BASE_IMAGE_REF}"
SCRIPT_EOF
[ "$?" = "0" ] || die "시험 스크립트 기록 실패"
ACT_SHA="$(sha256sum "$DIR/runner-push-test.sh" | cut -d' ' -f1)" || die "시험 스크립트 해시 계산 실패"
[ "$ACT_SHA" = "$EXPECT_SCRIPT_SHA" ] || die "시험 스크립트 sha256 불일치($ACT_SHA) — 중단(본체 실행 안 함)"
echo "OK: 시험 스크립트 sha256 일치 (리뷰·병합된 코드와 동일)"
CA_FILE="$HOME/.config/containers/certs.d/10.1.93.54/ca.crt"
[ "$(sha256sum "$CA_FILE" 2>/dev/null | cut -d' ' -f1)" = "$CA_SHA" ] || die "Podman CA trust 파일이 없거나 승인 Root와 다릅니다"
echo "OK: Podman CA trust(승인 Root) 확인"
export RUNNER_ROBOT_USER='robot$push'
IFS= read -rsp 'robot$push 비밀번호(화면에 표시되지 않음): ' RUNNER_ROBOT_SECRET < "$SECRET_SRC" || die "비밀번호 입력 실패(터미널 필요)"
echo
[ -n "$RUNNER_ROBOT_SECRET" ] || die "비밀번호가 비어 있습니다"
WRONG_ROBOT_SECRET="$(head -c 24 /dev/urandom | base64 | tr -d '/+=\n')x"
[ "$WRONG_ROBOT_SECRET" != "$RUNNER_ROBOT_SECRET" ] || die "음성시험용 문자열이 정상 비밀번호와 같습니다"
export RUNNER_ROBOT_SECRET WRONG_ROBOT_SECRET BASE_IMAGE_REF
{
  echo "# host=$(hostname) user=$(whoami) time=$(date -Is)"
  echo "RUN_ID=$RUN_ID  WRAPPER_SHA256=$WRAPPER_SHA256  SCRIPT_SHA256=$ACT_SHA  SOURCE=infra-harbor-trivy main 6edf816 (tree 25b8195fe1c1)"
  echo "BASE_IMAGE_REF=$BASE_IMAGE_REF"
  echo "CA_FILE_SHA256=$CA_SHA  tls_verify=on(default, --tls-verify=false 미사용)"
  echo "podman: $(podman --version 2>&1)"
} > "$LOG" || die "run.log 헤더 기록 실패"
# child rc와 tee rc를 각각 보존
bash "$DIR/runner-push-test.sh" 2>&1 | tee -a "$LOG"
PIPE_RC=("${PIPESTATUS[@]}")
CHILD_RC="${PIPE_RC[0]:-2}"; TEE_RC="${PIPE_RC[1]:-2}"
unset RUNNER_ROBOT_SECRET WRONG_ROBOT_SECRET
printf 'RUN_RC=%s TEE_RC=%s\n' "$CHILD_RC" "$TEE_RC" >> "$LOG" || die "run.log RUN_RC 기록 실패"
echo "--- child rc=$CHILD_RC tee rc=$TEE_RC"
[ "$TEE_RC" = "0" ] || { echo "RESULT=ERROR — 로그 기록(tee) 실패"; exit 2; }
seal() { ( cd "$DIR" && sha256sum "$@" > SHA256SUMS && sha256sum -c SHA256SUMS >/dev/null ) ; }
case "$CHILD_RC" in
  0) ;;
  1|2)
     [ ! -e "$DIR/review-run.env" ] && [ ! -e "$DIR/summary.txt" ] || { echo "RESULT=ERROR — 실패 run에 handoff 산출물이 존재"; exit 2; }
     if seal run.log runner-push-test.sh; then echo "RESULT=NOT_PASS (rc=$CHILD_RC) — handoff 미생성. 증적: $DIR"; else echo "RESULT=ERROR — 증적 seal 실패"; exit 2; fi
     exit "$CHILD_RC" ;;
  *) echo "RESULT=ERROR — 예상 밖 child rc=$CHILD_RC (원 rc는 run.log에 기록됨)"; seal run.log runner-push-test.sh >/dev/null 2>&1; exit 2 ;;
esac
TEST_TAG="$(grep -oE 'TEST_TAG=[A-Za-z0-9_][A-Za-z0-9._-]{0,127}' "$LOG" | tail -1 | cut -d= -f2)"
DIGEST="$(grep -oE 'MANIFEST_DIGEST=sha256:[0-9a-f]{64}' "$LOG" | tail -1 | cut -d= -f2)"
[ -n "$TEST_TAG" ] && [ -n "$DIGEST" ] || { echo "RESULT=ERROR — TEST_TAG/MANIFEST_DIGEST 파싱 실패"; exit 2; }
fail_handoff() { rm -f "$DIR/review-run.env" "$DIR/review-run.env.sha256" "$DIR/summary.txt" "$DIR/SHA256SUMS"; echo "RESULT=ERROR — $1 (handoff 폐기)"; exit 2; }
printf 'TEST_TAG=%s\nREVIEW_RUN_ID=%s\n' "$TEST_TAG" "$RUN_ID" > "$DIR/review-run.env" || fail_handoff "review-run.env 기록 실패"
( cd "$DIR" && sha256sum review-run.env > review-run.env.sha256 ) || fail_handoff "review-run.env.sha256 생성 실패"
{ echo "RUN_ID=$RUN_ID"; echo "TEST_TAG=$TEST_TAG"; echo "EXPECTED_HARBOR_DIGEST=$DIGEST"; echo "IMAGE=10.1.93.54/fds/test:$TEST_TAG"; echo "BASE_IMAGE_REF=$BASE_IMAGE_REF"; echo "WRAPPER_SHA256=$WRAPPER_SHA256"; echo "CHILD_RC=$CHILD_RC"; echo "EVIDENCE_SEAL=OK"; } > "$DIR/summary.txt" || fail_handoff "summary.txt 기록 실패"
seal run.log runner-push-test.sh review-run.env review-run.env.sha256 summary.txt || fail_handoff "최종 SHA256SUMS seal 실패"
cat "$DIR/SHA256SUMS"
echo "RESULT=PASS (AC-02 Runner Push/Pull/Negative; child rc=0, handoff 생성, evidence seal 완료)"
echo "결과 폴더: $DIR"
exit 0

WRAP_EOF
sha256sum ~/hbr-formal-run.sh
echo "9264f8a2227a783604b5a4593d43d375ce123b512a9479a79fa550efa9dfe311  $HOME/hbr-formal-run.sh" | sha256sum -c - && bash ~/hbr-formal-run.sh
